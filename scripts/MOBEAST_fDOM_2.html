fDOM_12PM_cobA <- fDOM_12PM_cobA %>%
  
  # removing outliers
  ## grouping by date, time and treatment
  group_by(date, time, treatment) %>% 
  ## caluculating the Interquantile range and of the raw data

         quant3 = quantile(fdom_conc_norm, probs = 0.75),
         IQR = quant3-quant1) %>% 
  ## removing data that are under Q1-1.5* or over 1.5+Q3*IQR, and labeling them outlier/not outliers

  ungroup() %>%

         quant3 = quantile(fdom_conc_norm, probs = 0.75),
         IQR = quant3-quant1) %>% 
  ## removing data that are under Q1-1.5* or over 1.5+Q3*IQR, and labeling them outlier/not outliers
  ## Important to filter later to understand why they are outliers. 
  mutate(outlier = ifelse( fdom_conc_norm > quant3 + (IQR*1.5) | fdom_conc_norm < quant1 - (IQR*1.5), "outlier", "not outlier")) %>% 
  ungroup() %>% 
  ## remove the outliers
